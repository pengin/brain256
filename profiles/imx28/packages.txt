# ImageBuilder rootfs に入れるパッケージ（1 行 1 件、「#」はコメント）。
# 通常の名前は追加し、「-name」は profile の既定パッケージから削除する。
# ImageBuilder の PACKAGES= へ渡す。

# Brain はルーターではないため、OpenWrt が既定で含める routing/firewall
# スタックを削除する。
-dnsmasq
-odhcpd-ipv6only
-firewall4
-nftables
-kmod-nft-offload
-ppp
-ppp-mod-pppoe

# M1 の USB-gadget-ether 経由 SSH 接続用に dropbear を残す。

# タイムゾーンデータ（検証済みの offline-opkg テストケースも兼ねる）。
zoneinfo-core
zoneinfo-asia

# タッチスクリーン（i2c0@0x48 の SX8650）を検証する。evtest で入力デバイス
#（/dev/input/eventN）を読み ABS_X/ABS_Y/BTN_TOUCH を表示し、GUI なしの CUI
# からパネルを確認できる。i2c-tools（i2cdetect）でバスをスキャンし、実際に
# ACK するチップを確認する。sx8650 ノードは未検証（linux-brain でも無効）で、
# probe は「reset failed」になったため、まず 0x48 が実装されているかを見る。
evtest
i2c-tools

# USB host-mode 診断。lsusb で hub とその先に挿した機器が、OTG host mode の
# usb0 上で実際に enumerate されることを確認する。
usbutils

# WLAN クライアント（GW-USEco300、Realtek RTL8192CU -> RTL8XXXU カーネル
# ドライバ）。wpa-supplicant-mini は WPA2-PSK の station mode 専用（EAP/TLS、
# AP/mesh なし）で、DHCP クライアントだけを使い AP スタックを持たないこの
# ビルドに合う。wireless-regdb は permissive な既定値ではなく国ごとの実際の
# チャネル／電力制限を cfg80211 に与え、iw は手動の scan/associate 診断用。
wpa-supplicant-mini
wireless-regdb
iw

# rtl8xxxu カーネルドライバが probe 時に request_firmware() する RTL8192CU
# firmware blob（rtlwifi/rtl8192cufw_TMSC.bin）。これがないと USB バス上では正常に
# enumerate されても probe が -11 で失敗し、dongle に wlan* インターフェースが現れない。
rtl8192cu-firmware

# オンボードオーディオ（SAIF 経由の SGTL5000 codec）を検証する。aplay/
# speaker-test で /proc/asound/cards に codec が現れ、PCM 再生が動くことを確認する。
alsa-utils

# コンテナ基盤 brainwrt-ct が使う ujail(procd 付属)。
procd-ujail
